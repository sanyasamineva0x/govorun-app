    private final class SequentialMockHTTPClient: HTTPClient, @unchecked Sendable {
        private let lock = NSLock()
        var responses: [(Data, URLResponse)] = []
        var errors: [Error?] = []
        private(set) var requests: [URLRequest] = []
        private var callIndex = 0

        func data(for request: URLRequest) async throws -> (Data, URLResponse) {
            lock.lock()
            requests.append(request)
            let index = callIndex
            callIndex += 1
            lock.unlock()

            if index < errors.count, let error = errors[index] {
                throw error
            }
            guard index < responses.count else {
                fatalError("SequentialMockHTTPClient: no response for call \(index)")
            }
            return responses[index]
        }
    }

           let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
           guard !trimmed.isEmpty else { return "" }

           let token: String
           do {
               token = try await authService.getAccessToken()
           } catch let error as AuthError {
               throw mapAuthError(error)
           } catch {
               if isCancellation(error) { throw CancellationError() }
               throw LLMError.networkError(error.localizedDescription)

               appName: hints.appName

               appName: hints.appName

           do {
