```swift
protocol HTTPClient: Sendable {
    func data(for request: URLRequest) async throws -> (Data, URLResponse)
}

extension URLSession: HTTPClient {}

final class MockHTTPClient: HTTPClient, @unchecked Sendable {
    private let lock = NSLock()
    var result: (Data, URLResponse)?
    var error: Error?
    private(set) var requests: [URLRequest] = []

    func data(for request: URLRequest) async throws -> (Data, URLResponse) {
        lock.lock()
        requests.append(request)
        lock.unlock()
        if let error { throw error }
        if let result { return result }
        let response = HTTPURLResponse(url: request.url ?? URL(string: "https://localhost")!, statusCode: 200, httpVersion: nil, headerFields: nil)!
        return (Data(), response as URLResponse)
    }
}
```

`getAccessToken() async throws -> String`:
1. Check cached token: `if let token = cachedToken, !isExpiringSoon(token) { return token.accessToken }` (synchronous, no await)
2. Check in-flight: `if let task = inFlightTask { return try await task.value }` (coalescing — Pitfall 3: defer must be inside Task body, NOT here)
3. Create new Task:
```swift
let task = Task<String, Error> {
    defer { self.inFlightTask = nil }
    guard let credentials = self.credentialProvider() else {
        throw AuthError.credentialsNotFound
    }
    let request = self.buildTokenRequest(
        clientId: credentials.clientId,
        clientSecret: credentials.secret,
        scope: self.scope
    )
    let (data, response): (Data, URLResponse)
    do {
        (data, response) = try await self.httpClient.data(for: request)
    } catch {
        throw AuthError.networkError(error.localizedDescription)
    }
    guard let httpResponse = response as? HTTPURLResponse else {
        throw AuthError.invalidResponse(statusCode: -1)
    }
    guard httpResponse.statusCode == 200 else {
        throw AuthError.invalidResponse(statusCode: httpResponse.statusCode)
    }
    let token = try self.parseTokenResponse(data)
    self.cachedToken = token
    return token.accessToken
}
inFlightTask = task
return try await task.value
```

`buildTokenRequest(clientId:clientSecret:scope:) -> URLRequest`:

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| App -> Sber OAuth endpoint | Credentials (clientId:secret) cross network boundary as Basic auth header over TLS |
| Keychain -> SberAuthService | Credentials read from Keychain via credentialProvider closure at token fetch time |
| Sber API -> App | OAuth response (access_token, expires_at) parsed from untrusted JSON |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-01 | Information Disclosure | credentialProvider closure | mitigate | Credentials read on-demand from Keychain, not stored as properties in SberAuthService. Closure is @Sendable, no capture of mutable state. |
| T-11-02 | Information Disclosure | cachedToken in memory | accept | In-process memory cache with 30-min token validity. Not persisted to disk. Acceptable risk for short-lived bearer tokens. |
| T-11-03 | Tampering | Sber OAuth endpoint | mitigate | TLS with certificate pinning via SberTrustPolicy (Phase 10). URLSession uses SberTrustDelegate for *.sberbank.ru domains. |
| T-11-04 | Denial of Service | Thundering herd on token refresh | mitigate | Actor coalescing via stored Task<String, Error>? ensures concurrent getAccessToken() calls produce exactly one HTTP request. |
| T-11-05 | Spoofing | Token replay | accept | Server-side concern. Tokens are short-lived (30 min). Out of scope for client. |
| T-11-06 | Input Validation | parseTokenResponse | mitigate | Explicit type checks: access_token must be String, expires_at must be TimeInterval. Malformed JSON throws AuthError.tokenParsingFailed. JSONSerialization (not string interpolation) prevents injection. |
| T-11-07 | Information Disclosure | Basic auth header | mitigate | Authorization header sent only over TLS (pinned). base64 encoding via Foundation Data.base64EncodedString() — standard, no custom crypto. |
</threat_model>
